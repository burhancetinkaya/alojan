cmake_minimum_required(VERSION 3.25)
project(alojan VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(ENV{PKG_CONFIG_PATH} "${CMAKE_SOURCE_DIR}/.local/pjsip/lib/pkgconfig:$ENV{PKG_CONFIG_PATH}")
find_package(PkgConfig REQUIRED)
pkg_check_modules(PJ REQUIRED libpjproject=2.16)
add_library(pjproject INTERFACE)
target_include_directories(pjproject SYSTEM INTERFACE ${PJ_INCLUDE_DIRS})
target_compile_options(pjproject INTERFACE ${PJ_CFLAGS_OTHER})
target_link_directories(pjproject INTERFACE ${PJ_STATIC_LIBRARY_DIRS})
if(APPLE)
  target_link_libraries(pjproject INTERFACE ${PJ_STATIC_LIBRARIES})
else()
  # Bundled Speex FFT calls back into PJ logging; resolve the static-library cycle.
  list(JOIN PJ_STATIC_LIBRARIES "," PJ_LINK_GROUP)
  target_link_libraries(pjproject INTERFACE "$<LINK_GROUP:RESCAN,${PJ_LINK_GROUP}>")
endif()
add_library(caller-identity STATIC services/voice-gateway/src/caller_identity.cpp)
target_include_directories(caller-identity PUBLIC services/voice-gateway/src)
target_link_libraries(caller-identity PUBLIC pjproject)
target_compile_options(caller-identity PRIVATE -Wall -Wextra -Wpedantic)
add_executable(voice-gateway services/voice-gateway/src/main.cpp)
target_compile_options(voice-gateway PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(voice-gateway PRIVATE caller-identity)
if(APPLE)
  # CMake otherwise deduplicates repeated -framework switches from pkg-config.
  target_link_options(pjproject INTERFACE "SHELL:-framework Foundation" "SHELL:-framework AppKit")
else()
  target_link_options(pjproject INTERFACE ${PJ_STATIC_LDFLAGS_OTHER})
endif()

target_compile_options(turn-detector PRIVATE -Wall -Wextra -Wpedantic)
if(ALOJAN_TURN_MODELS)
  find_library(ORT_LIBRARY NAMES onnxruntime libonnxruntime.so.1.31.0
    PATHS "${CMAKE_SOURCE_DIR}/.local/onnxruntime/lib" NO_DEFAULT_PATH REQUIRED)
  target_include_directories(turn-detector SYSTEM PRIVATE
    "${CMAKE_SOURCE_DIR}/.local/onnxruntime/include"
    "${CMAKE_SOURCE_DIR}/.local/src/pjproject-2.16/third_party/speex/libspeex"
    "${CMAKE_SOURCE_DIR}/.local/src/pjproject-2.16/third_party/speex/include")
  target_compile_definitions(turn-detector PRIVATE ALOJAN_TURN_MODELS FLOATING_POINT OUTSIDE_SPEEX)
  target_link_libraries(turn-detector PRIVATE "${ORT_LIBRARY}" pjproject)
endif()
